Kotlimo is a web application framework with expressive, elegant syntax. Routing, Kote views, an Eloquent-style ORM, sessions, CSRF, auth, validation, queues, mail, and Craft — on the JVM.

$ ./kotlimo new blog

Route.get("/posts", PostController::class, "index")

Route.post("/posts") { request ->
    Gate.authorize(Auth.user(), "create", Post())
    Post.create(validate(request.body, mapOf(
        "title" to "required|string|max:255"
    )))
    redirect("/posts")
}
Auth.attempt(mapOf(
    "email" to request.string("email"),
    "password" to request.string("password")
), remember = true)

val current = Auth.user() ?: return redirect("/login")
val token = Auth.createToken(current)
// Authorization: Bearer …
class Post : Model() {
    override val fillable = listOf("title", "body")
    override val softDeletes = true
    override val casts = mapOf("published" to "bool")

    companion object : ModelCompanion<Post>(Post::class)
}
